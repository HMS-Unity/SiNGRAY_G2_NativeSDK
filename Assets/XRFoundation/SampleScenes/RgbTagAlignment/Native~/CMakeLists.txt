cmake_minimum_required(VERSION 3.16)
project(singray_rgb_tag LANGUAGES CXX)
find_package(OpenCV 4.10 REQUIRED COMPONENTS core imgproc calib3d objdetect)
add_library(singray_rgb_tag SHARED rgb_tag.cpp)
target_compile_features(singray_rgb_tag PRIVATE cxx_std_17)
target_include_directories(singray_rgb_tag PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(singray_rgb_tag PRIVATE ${OpenCV_LIBS})
if(MSVC)
  set_property(TARGET singray_rgb_tag PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
